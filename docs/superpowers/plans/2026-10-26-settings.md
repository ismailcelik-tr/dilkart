```ts
// User settings: one synced user_settings row (plan section 16, Aşama 8).
import { SETTINGS_ID, Table, type LocalStore } from "../sync/localStore";
import type { Values } from "../sync/merge";

export enum Accent {
  US = "en-US",
  GB = "en-GB",
}

/** Stored for "no limit"; fits the smallint column. */
export const UNLIMITED = 9999;

export interface UserSettings {
  newPerDay: number;
  reviewsPerDay: number;
  dayStartHour: number;
  accent: Accent;
}

export const DEFAULT_SETTINGS: UserSettings = {
  newPerDay: 20,
  reviewsPerDay: 200,
  dayStartHour: 4,
  accent: Accent.US,
};

const COLUMN: Record<keyof UserSettings, string> = {
  newPerDay: "daily_new_limit",
  reviewsPerDay: "daily_review_limit",
  dayStartHour: "day_start_hour",
  accent: "tts_accent",
};

type Limit = "newPerDay" | "reviewsPerDay" | "dayStartHour";

export class Settings {
  constructor(private readonly store: LocalStore) {}

  /** Unset fields fall back to the defaults. */
  async get(): Promise<UserSettings> {
    const row = (await this.store.read(Table.Settings, SETTINGS_ID)) ?? {};
    const num = (k: Limit) => Number(row[COLUMN[k]] ?? DEFAULT_SETTINGS[k]);
    return {
      newPerDay: num("newPerDay"),
      reviewsPerDay: num("reviewsPerDay"),
      dayStartHour: num("dayStartHour"),
      accent: Object.values(Accent).find((a) => a === row[COLUMN.accent]) ?? DEFAULT_SETTINGS.accent,
    };
  }

  set(changes: Partial<UserSettings>): Promise<void> {
    const values: Values = {};
    for (const [k, v] of Object.entries(changes)) {
      values[COLUMN[k as keyof UserSettings]] = v;
    }
    return this.store.write(Table.Settings, SETTINGS_ID, values);
  }
}
```

```ts
/** Guest settings for the fields the account has not set. */
async function unsetSettings(guest: LocalStore, user: LocalStore): Promise<Values | undefined> {
  const theirs = (await guest.read(Table.Settings, SETTINGS_ID)) ?? {};
  const mine = (await user.read(Table.Settings, SETTINGS_ID)) ?? {};
  const fill = Object.entries(theirs).filter(([k, v]) => k !== "id" && (v ?? null) !== null && (mine[k] ?? null) === null);
  return fill.length === 0 ? undefined : Object.fromEntries(fill);
}
```

```ts
import { DEFAULT_SETTINGS, type Accent } from "@dilkart/core";
import type { Voice } from "expo-speech";
import { useEffect, useState } from "react";

import { englishVoices, pickVoice, say } from "@/drivers/speech";

import { useSettings } from "./library";

/** `available` is undefined until the voice list and the accent load. */
export function useSpeech() {
  const settings = useSettings();
  const [loaded, setLoaded] = useState<{ voices: Voice[]; accent: Accent }>();

  useEffect(() => {
    const voices = englishVoices().catch((): Voice[] => []);
    void Promise.all([voices, settings.get()]).then(([v, s]) => setLoaded({ voices: v, accent: s.accent }));
  }, [settings]);

  const accent = loaded?.accent ?? DEFAULT_SETTINGS.accent;
  const voice = loaded && pickVoice(loaded.voices, accent);
  return {
    available: loaded === undefined ? undefined : loaded.voices.length > 0,
    speak: (text: string) => say(text, accent, voice),
  };
}
```

```ts
export type Field = keyof UserSettings;

interface Choice {
  value: UserSettings[Field];
  label: string;
}

const numbers = (values: number[]): Choice[] => values.map((v) => ({ value: v, label: String(v) }));
const hour = (h: number) => `${String(h).padStart(2, "0")}:00`;

export const FIELDS: Record<Field, { title: string; choices: Choice[] }> = {
  newPerDay: { title: "Günlük yeni kart", choices: numbers([5, 10, 20, 30, 50]) },
  reviewsPerDay: {
    title: "Günlük tekrar",
    choices: [...numbers([50, 100, 200, 500]), { value: UNLIMITED, label: "Sınırsız" }],
  },
  dayStartHour: { title: "Gün başlangıcı", choices: [0, 1, 2, 3, 4, 5, 6].map((h) => ({ value: h, label: hour(h) })) },
  accent: {
    title: "Aksan",
    choices: [
      { value: Accent.US, label: "ABD" },
      { value: Accent.GB, label: "İngiliz" },
    ],
  },
};

/** A value synced from elsewhere may be off the list; it shows as is. */
export const labelOf = (field: Field, value: UserSettings[Field]) =>
  FIELDS[field].choices.find((c) => c.value === value)?.label ?? String(value);

export const isField = (s: string | undefined): s is Field => s !== undefined && s in FIELDS;
```

```tsx
  const settings = useSettings();
  const failedCount = useFailedCount();
  const [data] = useLoad(
    useCallback(async () => ({ values: await settings.get(), failed: await failedCount() }), [settings, failedCount]),
  );
  const dark = useColorScheme() === "dark";
  if (data === undefined) {
    return null;
  }

  const row = (field: Field) => (
    <Row
      key={field}
      title={FIELDS[field].title}
      subtitle={labelOf(field, data.values[field])}
      onPress={() => router.push({ pathname: "/settings/choose", params: { field } })}
    />
  );

  return (
    <Screen>
      <Group title="Çalışma">{STUDY.map((f) => row(f))}</Group>
      <Group title="Ses">{row("accent")}</Group>
      <Group title="Görünüm">
        <SwitchRow icon="moon.fill" title="Koyu görünüm" value={dark} onChange={(on) => setScheme(on ? "dark" : "light")} />
      </Group>
      {data.failed > 0 && <ErrorText text={`Gönderilemeyen değişiklik: ${data.failed}`} />}
    </Screen>
  );
}

/** App-wide, native header and keyboard included. */
function setScheme(scheme: Scheme) {
  Appearance.setColorScheme(scheme);
  void rememberScheme(scheme);
}
```

```tsx
import type { UserSettings } from "@dilkart/core";
import { Stack } from "expo-router";
import { useCallback } from "react";

import { useLoad, useSettings } from "@/services/library";
import { Group, Row, Screen } from "@/ui/components";
import { goBack } from "@/ui/navigation";

import { FIELDS, type Field } from "./choices";

export function ChooseScreen({ field }: { field: Field }) {
  const settings = useSettings();
  const [values] = useLoad(useCallback(() => settings.get(), [settings]));
  if (values === undefined) {
    return null;
  }

  const pick = async (value: UserSettings[Field]) => {
    await settings.set({ [field]: value } as Partial<UserSettings>);
    goBack("/settings");
  };

  return (
    <Screen>
      <Stack.Screen options={{ title: FIELDS[field].title }} />
      <Group>
        {FIELDS[field].choices.map((c) => (
          <Row key={String(c.value)} title={c.label} selected={c.value === values[field]} onPress={() => pick(c.value)} />
        ))}
      </Group>
    </Screen>
  );
}
```
