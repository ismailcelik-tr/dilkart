# İstatistik sayfası ve seri satırı: uygulama planı

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ana sayfada seri + günlük hedef satırı, ayrı İstatistik sayfası ve Ayarlar'da günlük hedef.

**Architecture:** Core'da `Stats` repository (`today`, `report`). `today` ana sayfa için `LocalStore.answersPerDay` dar sorgusunu kullanır; `report` tüm logları okuyup mevcut saf fonksiyonları çağırır. Hedef yeni `user_settings.daily_goal` sütununda (yerel + sunucu migration). Ekranlar `useStats` üzerinden erişir; grafikler düz `View`.

**Tech Stack:** TypeScript, Vitest, Supabase (pgTAP, sync harness), Expo Router + jest-expo + RNTL.

Spec: `docs/superpowers/specs/2026-10-27-stats-design.md`

## Global Constraints

- Kapsam tüm desteler. Seri: ≥1 cevaplı çalışma günü; bugün çalışılmadıysa dünkü seri gün bitene kadar sürer.
- Cevap = yeni + tekrar + pratik (her `review_logs` satırı).
- Hedef seçenekleri `Kapalı · 20 · 50 · 100 · 200`; `0` = kapalı (`GOAL_OFF`), `null` → 50.
- Gün: `dayStartHour` + cihazın şu anki saat dilimi; `tz_offset_min` kullanılmaz.

```ts
/** Consecutive study days; today's streak survives until today ends. */
export function streak(reviews: Review[], now: string, ctx: DayContext): { current: number; longest: number } {
  const studied = [...new Set(reviews.map((r) => day(r.reviewedAt, ctx)))].sort((a, b) => a - b);
  return streakOfDays(studied, day(now, ctx));
}

/** `streak` on ascending, distinct study days. */
export function streakOfDays(days: number[], today: number): { current: number; longest: number } {
  let longest = 0;
  let run = 0;
  for (const [i, d] of days.entries()) {
    run = i > 0 && d - (days[i - 1] ?? 0) === 1 ? run + 1 : 1;
    longest = Math.max(longest, run);
  }

  const last = days.at(-1);
  const current = last !== undefined && today - last <= 1 ? run : 0;
  return { current, longest };
}

/** Weekday of a study day; 0 = Sunday. */
export const weekday = (day: number) => (day + EPOCH_WEEKDAY) % DAYS_PER_WEEK;
```

```ts
// Home streak row and statistics page; docs/superpowers/specs/2026-10-27-stats-design.md.
import { dayKey, deviceOffset } from "../fsrs/day";
import {
  dailyCounts,
  forecast,
  retention,
  stateCounts,
  streak,
  streakOfDays,
  type Retention,
  type StateCounts,
} from "../stats/stats";
import type { LocalStore } from "../sync/localStore";
import type { Library } from "./library";
import { reviewsByCard } from "./reviews";
import type { Settings } from "./settings";

const WEEK = 7;
const MONTH = 30;

export interface Today {
  streak: number;
  answers: number;
  /** GOAL_OFF when off. */
  goal: number;
}

export interface Report {
  /** Today's study day number, for weekday labels. */
  today: number;
  streak: { current: number; longest: number };
  /** Answers per day, last 30 days, oldest first. */
  daily: number[];
  goal: number;
  retention7: Retention | null;
  retention30: Retention | null;
  states: StateCounts;
  /** Due cards per day, next 7 days from today. */
  forecast: number[];
}

export class Stats {
  constructor(
    private readonly store: LocalStore,
    private readonly library: Library,
    private readonly settings: Settings,
    private readonly now: () => number = Date.now,
    private readonly tzOffsetMin: () => number = deviceOffset,
  ) {}

  /** Cheap enough for every home screen load: one grouped query, no log rows. */
  async today(): Promise<Today> {
    const { dayStartHour, dailyGoal } = await this.settings.get();
    const tz = this.tzOffsetMin();
    const perDay = await this.store.answersPerDay(dayStartHour, tz);
    const today = dayKey(new Date(this.now()), dayStartHour, tz);
    return {
      streak: streakOfDays(perDay.map((d) => d.day), today).current,
      answers: perDay.find((d) => d.day === today)?.count ?? 0,
      goal: dailyGoal,
    };
  }

  async report(): Promise<Report> {
    const { dayStartHour, dailyGoal } = await this.settings.get();
    const ctx = { dayStartHour, tzOffsetMin: this.tzOffsetMin() };
    const now = new Date(this.now()).toISOString();
    const byCard = [...(await reviewsByCard(this.store)).values()];
    const reviews = byCard.flat();
    const states = (await this.library.allCards()).map((c) => c.fsrs);
    return {
      today: dayKey(now, dayStartHour, ctx.tzOffsetMin),
      streak: streak(reviews, now, ctx),
      daily: dailyCounts(reviews, now, ctx),
      goal: dailyGoal,
```

```tsx
/** Home row: streak, today's answers and the goal bar; opens the statistics page. */
export function StreakRow({ today }: { today: Today }) {
  const { streak, answers, goal } = today;
  const hasGoal = goal !== GOAL_OFF;
  return (
    <Group>
      <Row
        title={streak > 0 ? `🔥 ${streak} gün seri` : "Seriyi başlat"}
        subtitle={subtitle(today)}
        onPress={() => router.push("/stats")}
      >
        {hasGoal && <Progress ratio={Math.min(answers / goal, 1)} done={answers >= goal} />}
      </Row>
    </Group>
  );
}

function subtitle({ streak, answers, goal }: Today): string {
  const hasGoal = goal !== GOAL_OFF;
  const count = hasGoal ? `${answers}/${goal}` : String(answers);
  if (hasGoal && answers >= goal) {
    return `Hedef tamam: ${count} ✓`;
  }
  if (streak > 0 && answers === 0) {
    const next = `Bugün çalışırsan ${streak + 1} olur`;
    return hasGoal ? `${next} · ${count}` : next;
  }
  return `Bugün ${count} cevap`;
}

function Progress({ ratio, done }: { ratio: number; done: boolean }) {
  const theme = useTheme();
  return (
    <View testID="goal-progress" style={[styles.track, { backgroundColor: theme.separator }]}>
      <View style={[styles.fill, { width: `${ratio * 100}%`, backgroundColor: done ? theme.success : theme.tint }]} />
    </View>
  );
}

const styles = StyleSheet.create({
  track: { height: 4, borderRadius: 2, marginTop: 6, overflow: "hidden" },
  fill: { height: "100%" },
});
```

```tsx
  if (report === undefined) {
    return null;
  }

  return (

      </Group>

function History({ report }: { report: Report }) {
  const theme = useTheme();
  const total = report.daily.reduce((a, b) => a + b, 0);
  const average = Math.round(total / report.daily.length);
  const most = Math.max(...report.daily);
  const caption = [styles.caption, { color: theme.secondary }];

  return (
    <>
      <Group title="Seri">
        <InfoRow title="Şu an" value={`${report.streak.current} gün`} />

      </Group>

const rate = (r: Retention | null) => (r === null ? "—" : `%${Math.round(r.rate * 100)} · ${r.total} cevap`);

  if (offset === 1) {
    return "Yarın";
  }
  return WEEKDAYS[weekday(today + offset)] ?? "";
```
